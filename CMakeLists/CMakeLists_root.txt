# 3.24 is required for CMAKE_DEPENDS_USE_COMPILER, which uses the compiler
# for Fortran module dependency scanning instead of the legacy makedepend approach.

set(CMAKE_POLICY_DEFAULT_CMP0048 NEW)

  cmake_policy(SET CMP0144 NEW)
endif()

if (${CMAKE_SOURCE_DIR} STREQUAL ${CMAKE_BINARY_DIR})
  message(FATAL_ERROR "Out-of-source build required. Run cmake from a directory different from the source directory.")
endif()

# set default flags. Must be called before project() enables languages.
macro(set_mode_flags languages mode value)
  foreach(lang ${languages})
    set(CMAKE_${lang}_FLAGS_${mode} "${value}" CACHE STRING "")
  endforeach()
endmacro()

set_mode_flags("C;CXX" "DEBUG" "-g")
set_mode_flags("C;CXX" "RELWITHDEBINFO" "-g -DNDEBUG")
set_mode_flags("C;CXX" "RELEASE" "-DNDEBUG")
set_mode_flags("C;CXX" "MINSIZEREL" "-DNDEBUG")

set_mode_flags("Fortran" "DEBUG" "-g")
set_mode_flags("Fortran" "RELWITHDEBINFO" "-g")
set_mode_flags("Fortran" "RELEASE" "")
set_mode_flags("Fortran" "MINSIZEREL" "")

# get library dirs for external MPI enabled libraries, like scalapack
set (MPI_IS_OMPI FALSE)
if(NOT CMAKE_Fortran_COMPILER_ID STREQUAL "Cray")
  execute_process(
    COMMAND ${MPI_Fortran_COMPILER} --showme:libdirs
    OUTPUT_VARIABLE _MPI_LIBDIRS
    RESULT_VARIABLE _SHOWME_RC
    OUTPUT_STRIP_TRAILING_WHITESPACE
    ERROR_QUIET
  )
  if(NOT _SHOWME_RC EQUAL 0 OR _MPI_LIBDIRS STREQUAL "")
    execute_process(
      COMMAND ${MPI_Fortran_COMPILER} -show
      OUTPUT_VARIABLE _MPI_SHOW
      OUTPUT_STRIP_TRAILING_WHITESPACE
    )
    string(REGEX MATCHALL "-L[^ ]+" _MPI_LIBDIR_FLAGS "${_MPI_SHOW}")
    string(REPLACE "-L" "" _MPI_LIBDIRS "${_MPI_LIBDIR_FLAGS}")
  endif()

  # for scaLAPACK it is useful to detect if we are using OMPI
  execute_process(COMMAND grep -i "OMPI_MPI" ${MPI_Fortran_F77_HEADER_DIR}/mpi.h
                  RESULT_VARIABLE MPI_GREP_RESULT
                  OUTPUT_QUIET
                  ERROR_QUIET)
  if(MPI_GREP_RESULT EQUAL 0)
    set(MPI_IS_OMPI TRUE)
  endif()
endif()

if(NOT VASP_NVPL_MATH AND CMAKE_Fortran_COMPILER_ID STREQUAL "NVHPC")
  if(DEFINED ENV{MKLROOT} OR DEFINED ENV{ONEAPI_ROOT})
    if(VASP_OPENMP)
      message(STATUS "NVHPC compiler detected with MKL environment, using -Mmkl (threaded)")
    else()
      message(STATUS "NVHPC compiler detected with MKL environment, using -Mmkl (sequential)")
    endif()
    add_library(BLAS::BLAS INTERFACE IMPORTED)
    add_library(LAPACK::LAPACK INTERFACE IMPORTED)
    # If OpenMP is enabled, link OpenMP to MKL to get threaded variant
    if(VASP_OPENMP)
      target_link_options(BLAS::BLAS INTERFACE "-Mmkl" "-mp")
      target_link_options(LAPACK::LAPACK INTERFACE "-Mmkl" "-mp")
    else()
      # link against sequential variant
      target_link_options(BLAS::BLAS INTERFACE "-Mmkl")
      target_link_options(LAPACK::LAPACK INTERFACE "-Mmkl")
    endif()
    list(APPEND VASP_EXTERNAL_LIBS BLAS::BLAS LAPACK::LAPACK)
    set(VASP_NVHPC_MKL TRUE)
  endif()
endif()

# Find CUDA
message(STATUS "-------- GPU support detection ----------")

# AUTO must stay distinct from an explicit OFF, or detection overrides the user.
if(VASP_CUDA STREQUAL "AUTO")
  set(_vasp_cuda_request AUTO)
elseif(VASP_CUDA)
  set(_vasp_cuda_request ON)
else()
  set(_vasp_cuda_request OFF)
endif()

if(_vasp_cuda_request STREQUAL "ON" AND VASP_OMP_OFFLOAD)
  message(FATAL_ERROR "VASP_CUDA=ON and VASP_OMP_OFFLOAD are mutually exclusive. Please select only one.")
endif()

if(_vasp_cuda_request STREQUAL "OFF" OR VASP_OMP_OFFLOAD)
  set(VASP_CUDA OFF)
  if(_vasp_cuda_request STREQUAL "OFF")
    # USENCCL without NVCUDA does not compile; VASP_OMP_OFFLOAD has its own path.
    set(VASP_USE_NCCL OFF)
    message(STATUS "VASP_CUDA=OFF requested - OpenACC support disabled.")
  endif()
else()
  include(CheckLanguage)
  check_language(CUDA)
  if(CMAKE_CUDA_COMPILER)
    enable_language(CUDA)
    set(VASP_CUDA ON)
  elseif(_vasp_cuda_request STREQUAL "ON")
    message(FATAL_ERROR "VASP_CUDA=ON was requested but no CUDA compiler was found.")
  else()
    set(VASP_CUDA OFF)
    set(VASP_USE_NCCL OFF)
    message(STATUS "No CUDA capable compiler found - OpenACC support disabled.")
  endif()
endif()
unset(_vasp_cuda_request)

if(VASP_CUDA)
  message(STATUS "CUDA and an offloading device was found enabling OpenACC port")

  if("${CMAKE_CUDA_ARCHITECTURES}" STREQUAL "native")
    if(VASP_LIBXC)
      message(STATUS "Disabling LibXC support as it is incompatible with NVIDIA OpenACC port")
      set(VASP_LIBXC OFF)
    endif()
    message(STATUS "setting default CUDA compute capability of present device")
    set(VASP_GPU_FLAG "-gpu=ccnative")
    message(STATUS "overwrite this by calling cmake with -DCMAKE_CUDA_ARCHITECTURES=XX")
  else()
    message(STATUS "setting user specified CUDA compute capability")
    # Convert comma-separated to semicolon-separated list
    string(REPLACE "," ";" _vasp_cuda_arch_list "${CMAKE_CUDA_ARCHITECTURES}")
    list(POP_FRONT _vasp_cuda_arch_list VASP_FIRST_ARCH)
    set(VASP_GPU_FLAG "-gpu=cc${VASP_FIRST_ARCH}")

    foreach(arch IN LISTS _vasp_cuda_arch_list)
      set(VASP_GPU_FLAG "${VASP_GPU_FLAG},cc${arch}")
    endforeach()
  endif()

  if(DEFINED ENV{NVROOT})
    set(NVHPC_INCLUDE "$ENV{NVROOT}/compilers/include")
  endif()

  if(VASP_CUDA_VERSION AND NOT "${VASP_CUDA_VERSION}" STREQUAL "Default")
    message(STATUS "CUDA version: ${VASP_CUDA_VERSION}")
    set(VASP_GPU_FLAG "${VASP_GPU_FLAG},cuda${VASP_CUDA_VERSION}")
  else()
    message(STATUS "default CUDA version, use -DVASP_CUDA_VERSION=XX to change")
  endif()
  message(STATUS "CUDA compile flags: ${VASP_GPU_FLAG}")

  # NVHPC installations use different environment variables depending on module
  # setup (e.g., NVHPC vs NVHPC_ROOT) and may place libraries in lib/lib64
  # subfolders. Provide a broader set of hints for find_library().
  set(_VASP_NVHPC_ROOTS)
  foreach(_var NVHPC NVHPC_ROOT)
    if(DEFINED ENV{${_var}} AND NOT "$ENV{${_var}}" STREQUAL "")
      list(APPEND _VASP_NVHPC_ROOTS "$ENV{${_var}}")
    endif()
  endforeach()
  list(REMOVE_DUPLICATES _VASP_NVHPC_ROOTS)

  set(_VASP_NVHPC_MATH_LIB_DIRS)
  set(_VASP_NVHPC_NCCL_LIB_DIRS)
  set(_VASP_NVHPC_NVSHMEM_LIB_DIRS)
  foreach(_root IN LISTS _VASP_NVHPC_ROOTS)
    list(APPEND _VASP_NVHPC_MATH_LIB_DIRS
      "${_root}/math_libs"
      "${_root}/math_libs/lib"
      "${_root}/math_libs/lib64"
    )
    list(APPEND _VASP_NVHPC_NCCL_LIB_DIRS
      "${_root}/comm_libs/nccl"
      "${_root}/comm_libs/nccl/lib"
      "${_root}/comm_libs/nccl/lib64"
    )
    list(APPEND _VASP_NVHPC_NVSHMEM_LIB_DIRS
      "${_root}"
      "${_root}/lib"
      "${_root}/lib64"
      "${_root}/comm_libs/nvshmem"
      "${_root}/comm_libs/nvshmem/lib"
      "${_root}/comm_libs/nvshmem/lib64"
    )
  endforeach()
  list(REMOVE_DUPLICATES _VASP_NVHPC_MATH_LIB_DIRS)
  list(REMOVE_DUPLICATES _VASP_NVHPC_NCCL_LIB_DIRS)
  list(REMOVE_DUPLICATES _VASP_NVHPC_NVSHMEM_LIB_DIRS)

  set(_CUDA_LIB_LIST "cublas,cusolver,cufft,nccl")
  # find the library to be able to add its path to runpath / rpath in the binary
  find_library(CUBLAS_LIBRARY
    NAMES cublas libcublas.so
    HINTS ${_VASP_NVHPC_MATH_LIB_DIRS}
  )
  if(CUBLAS_LIBRARY)
    get_filename_component(lib_dir ${CUBLAS_LIBRARY} DIRECTORY)
    list(APPEND VASP_EXTERNAL_LIB_DIRS ${lib_dir})
  endif()

  find_library(NCCL_LIBRARY
    NAMES nncl libnccl.so libnccl.so.2
    HINTS ${_VASP_NVHPC_NCCL_LIB_DIRS}
  )
  if(NCCL_LIBRARY)
    get_filename_component(lib_dir ${NCCL_LIBRARY} DIRECTORY)
    list(APPEND VASP_EXTERNAL_LIB_DIRS ${lib_dir})
  endif()

  find_library(NVSHMEM_LIBRARY
    NAMES nvshmem_host libnvshmem_host.so libnvshmem_host.so.3
    HINTS ${_VASP_NVHPC_NVSHMEM_LIB_DIRS}
  )
  if(NVSHMEM_LIBRARY)
    get_filename_component(lib_dir ${NVSHMEM_LIBRARY} DIRECTORY)
    list(APPEND VASP_EXTERNAL_LIB_DIRS ${lib_dir})
  endif()

  # we cannot use cusolvermp without scalapack
  if(NOT VASP_SCALAPACK)

# --- helper function to convert target include dirs to -I flags ---
function(target_include_dirs_to_flags target out_var)
    if(TARGET ${target})
        get_target_property(dirs ${target} INTERFACE_INCLUDE_DIRECTORIES)
        if(dirs)
            set(flags "")
            foreach(d ${dirs})
                list(APPEND flags "-I${d}")
            endforeach()
            set(${out_var} "${flags}" PARENT_SCOPE)
        else()
            set(${out_var} "" PARENT_SCOPE)
        endif()
    else()
        # target does not exist, just return empty
        set(${out_var} "" PARENT_SCOPE)
    endif()
endfunction()

    list(APPEND PREPROCESSOR_INCLUDE_FLAGS ${tmp_flags})
endforeach()
